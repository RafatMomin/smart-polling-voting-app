 package onetoone.Users.service;
 
 import org.springframework.mail.SimpleMailMessage;
 import org.springframework.mail.javamail.JavaMailSender;
 import org.springframework.stereotype.Service;
 
 @Service
 public class MailingService {
 
     private final JavaMailSender mailSender;
     public MailingService(JavaMailSender mailSender) {
         this.mailSender = mailSender;
     }
 
     // email for verifying account when a user signs up
     public void sendVerificationEmail(String to, String verificationLink)
     {
         SimpleMailMessage message = new SimpleMailMessage();
         message.setTo(to);
 
         message.setSubject("Welcome to Voting and Polling App - Verify Your Email");
         String emailBody = String.format(
             "Welcome to Voting and Polling App! \n\n" +
             "Thank you for creating an account with us. To complete your " +
             "registration and start using our platform, please verify your email address. \n\n" +
             "Your verification token is: \n" + verificationLink + "\n\n" +
             "Please use this token to verify your account through our application.\n\n" +
             "Best regards, \n" +
             "The Polling and Voting Team\n\n" +
             "----------------------------------------------------------\n" +
             "Need help? Contact us at votingandpolling@gmail.com"
         );
 
         message.setText(emailBody);
         mailSender.send(message);
     }
 
     // email for verifying an account when a user wants to reset their password
     public void sendPasswordResetEmail(String to, String resetToken)
     {
         SimpleMailMessage message = new SimpleMailMessage();
         message.setTo(to);
 
         message.setSubject("Voting and Polling App - Password Reset Request");
         String emailBody = String.format(
             "Hello,\n\n" +
             "We received a request to reset your password for your Voting and Polling App account.\n\n" +
             "Your password reset token is: \n" + resetToken + "\n\n" +
             "Please use this token to reset your password through our application.\n\n" +
             "Best regards,\n" +
             "The Voting and Polling Team\n\n" +
             "-----------------------------------------------------------\n" +
             "Need help? Contact us at votingandpolling@gmail.com"
         );
 
         message.setText(emailBody);
         mailSender.send(message);
     }
 
     // email to admin with access code
     public void sendPollAccessCodeEmail(String to, String pollTitle, String accessCode)
     {
         SimpleMailMessage message = new SimpleMailMessage();
         message.setTo(to);
 
         message.setSubject("Your Private Poll Access Code - " + pollTitle);
         String emailBody = String.format(
                 "Hello,\n\n" +
                         "You have successfully created a private poll: \"" + pollTitle + "\"\n\n" +
                         "Your access code is: " + accessCode + "\n\n" +
                         "Share this code with users who you want to join your poll. " +
                         "They can use this code to join and participate in your poll.\n\n" +
                         "Important: Keep this code secure. Only share it with users you want to grant access to.\n\n" +
                         "Best regards,\n" +
                         "The Voting and Polling Team\n\n" +
                         "-----------------------------------------------------------\n" +
                         "Need help? Contact us at votingandpolling@gmail.com"
         );
 
         message.setText(emailBody);
         mailSender.send(message);
     }
 }
